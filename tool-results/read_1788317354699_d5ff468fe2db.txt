   112→// Mini bar chart (CSS-only, no chart library needed)
   113→// ============================================================
   114→function MiniBarChart({ data, height = 80 }: { data: ChartPoint[]; height?: number }) {
   115→  const maxCount = Math.max(...data.map((d) => d.count), 1);
   116→
   117→  return (
   118→    <div className="flex items-end gap-[3px] w-full" style={{ height }}>
   119→      {data.map((d, i) => {
   120→        const barH = (d.count / maxCount) * 100;
   121→        const isToday = d.date === new Date().toISOString().split('T')[0];
   122→        return (
   123→          <div
   124→            key={d.date}
   125→            className="flex-1 min-w-0 flex flex-col items-center gap-1 group relative"
   126→          >
   127→            {/* Tooltip */}
   128→            <div className="absolute -top-8 left-1/2 -translate-x-1/2 hidden group-hover:flex items-center justify-center bg-popover text-popover-foreground text-[10px] px-1.5 py-0.5 rounded shadow-md border whitespace-nowrap z-10 pointer-events-none">
   129→              {format(parseISO(d.date), "dd MMM", { locale: ptBR })}: {d.count} lead{d.count !== 1 ? 's' : ''}
   130→            </div>
   131→            <div
   132→              className={`w-full rounded-sm transition-all duration-300 ${
   133→                isToday
   134→                  ? 'bg-emerald-500'
   135→                  : d.count > 0
   136→                    ? 'bg-blue-400/70 dark:bg-blue-500/60'
   137→                    : 'bg-muted'
   138→              }`}
   139→              style={{ height: `${Math.max(barH, 2)}%` }}
   140→            />
   141→          </div>
   142→        );
   143→      })}
   144→    </div>
   145→  );
   146→}

   148→// ============================================================
   149→// Funnel Visualization
   150→// ============================================================
   151→function FunnelVisualization({ byStage }: { byStage: Record<string, number> }) {
   152→  const funnelStages = ['LEAD', 'PROSPECT', 'VISITA', 'NEGOCIACAO', 'PROPOSTA', 'FECHADO'];
   153→  const data = funnelStages
   154→    .map((s) => ({ stage: s, count: byStage[s] || 0 }))
   155→    .filter((d) => d.count > 0);
   156→
   157→  const maxCount = Math.max(...data.map((d) => d.count), 1);
   158→
   159→  if (data.length === 0) {
   160→    return (
   161→      <div className="text-center py-8 text-muted-foreground text-sm">
   162→        Nenhum dado de funil disponível
   163→      </div>
   164→    );
   165→  }
   166→
   167→  return (
   168→    <div className="space-y-2">
   169→      {data.map((d, i) => {
   170→        const widthPct = Math.max((d.count / maxCount) * 100, 15);
   171→        const config = STAGE_CONFIG[d.stage] || { label: d.stage, color: 'text-gray-600', bg: 'bg-gray-100 dark:bg-gray-800' };
   172→        const prevCount = i > 0 ? data[i - 1].count : d.count;
   173→        const dropRate = prevCount > 0 ? (((prevCount - d.count) / prevCount) * 100).toFixed(0) : '0';
   174→
   175→        return (
   176→          <div key={d.stage} className="flex items-center gap-3">
   177→            <div className="w-24 text-right text-[11px] font-medium text-muted-foreground flex-shrink-0">
   178→              {config.label}
   179→            </div>
   180→            <div className="flex-1 relative">
   181→              <div
   182→                className={`${config.bg} rounded-sm h-7 flex items-center px-2 transition-all duration-500`}
   183→                style={{ width: `${widthPct}%` }}
   184→              >
   185→                <span className={`text-[11px] font-bold ${config.color}`}>{d.count}</span>
   186→              </div>
   187→            </div>
   188→            {i > 0 && d.count < prevCount && (
   189→              <div className="w-16 text-right flex-shrink-0">
   190→                <span className="text-[10px] text-red-500 font-medium">-{dropRate}%</span>
   191→              </div>
   192→            )}
   193→          </div>
   194→        );
   195→      })}
   196→    </div>
   197→  );
   198→}
