   120→  // Salvar perfil do usuário
   121→  async function saveProfile() {
   122→    if (!userName.trim()) {
   123→      toast.error('Nome é obrigatório');
   124→      return;
   125→    }
   126→
   127→    setSavingProfile(true);
   128→    try {
   129→      const res = await fetch('/api/profile', {
   130→        method: 'PUT',
   131→        headers: { 'Content-Type': 'application/json' },
   132→        body: JSON.stringify({ name: userName.trim(), phone: userPhone }),
   133→      });
   134→
   135→      if (res.ok) {
   136→        toast.success('Perfil atualizado com sucesso!');
   137→        // Atualizar a sessão para refletir o novo nome na sidebar
   138→        await updateSession({ name: userName.trim() });
   139→      } else {
   140→        const data = await res.json();
   141→        throw new Error(data.error || 'Erro ao salvar');
   142→      }
   143→    } catch {
   144→      toast.error('Erro ao atualizar perfil');
   145→    } finally {
   146→      setSavingProfile(false);
   147→    }
   148→  }

   160→      toast.error('Erro ao iniciar conexão com Google Calendar');
   161→      setGcConnecting(false);

   163→  }

   165→  async function disconnectGoogleCalendar() {
   166→    setGcDisconnecting(true);
   167→    try {
   168→      const res = await fetch('/api/google-calendar/disconnect', { method: 'POST' });
   169→      if (res.ok) {
   170→        setGcConnected(false);
   171→        toast.success('Google Calendar desconectado');
   172→      } else {
   173→        throw new Error();
   174→      }
   175→    } catch {
   176→      toast.error('Erro ao desconectar Google Calendar');
   177→    } finally {
   178→      setGcDisconnecting(false);
   179→    }
   180→  }

   182→  async function saveTelegramChatId() {
   183→    if (!tgChatId.trim()) {
   184→      toast.error('Insira o Chat ID');
   185→      return;
   186→    }
   187→    setTgSaving(true);
   188→    try {
   189→      const res = await fetch('/api/settings/telegram', {
   190→        method: 'PUT',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify({ action: 'connect', chatId: tgChatId.trim() }),
   193→      });
   194→      const data = await res.json();
   195→      if (res.ok) {
   196→        setTgConnected(true);
   197→        toast.success('Telegram vinculado com sucesso!');
   198→      } else {
   199→        toast.error(data.error || 'Erro ao vincular Telegram');

   201→    } catch {
   202→      toast.error('Erro ao vincular Telegram');

   204→      setTgSaving(false);

   208→  async function disconnectTelegram() {

   211→      const res = await fetch('/api/settings/telegram', {
   212→        method: 'PUT',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({ action: 'disconnect' }),

   248→  function addPollFormId() {

   256→  function updatePollFormId(index: number, value: string) {
   257→    const updated = [...pollFormIds];
   258→    updated[index] = value.replace(/\D/g, ''); // apenas dígitos
   259→    setPollFormIds(updated);

   262→  async function savePollConfig() {
   263→    setPollSaving(true);
   264→    try {
   265→      const validIds = pollFormIds.filter((id) => id.length > 0);
   266→      const res = await fetch('/api/cron/fetch-meta-leads/config', {
   267→        method: 'PUT',
   268→        headers: { 'Content-Type': 'application/json' },
   269→        body: JSON.stringify({ enabled: pollEnabled, formIds: validIds }),
   270→      });
   271→      const data = await res.json();
   272→      if (res.ok) {
   273→        toast.success(pollEnabled ? 'Polling ativado! Leads serão importados a cada 5 minutos.' : 'Polling desativado.');
   274→        setPollFormIds(validIds.length ? validIds : ['']);
   275→        setPollSavedEnabled(pollEnabled); // sincronizar estado salvo
   276→      } else {
   277→        toast.error(data.error || 'Erro ao salvar configuração');
   278→      }
   279→    } catch {
   280→      toast.error('Erro ao salvar configuração do polling');
   281→    } finally {
   282→      setPollSaving(false);
   283→    }
   284→  }

   286→  async function triggerPollNow() {
   287→    // Se há mudanças não salvas (enabled ou formIds divergem do banco), salvar primeiro
   288→    const hasUnsaved = pollEnabled !== pollSavedEnabled;
   289→    if (hasUnsaved) {
   290→      toast.info('Salvando configuração antes de executar...');
   291→      const validIds = pollFormIds.filter((id) => id.length > 0);
   292→      try {
   293→        const saveRes = await fetch('/api/cron/fetch-meta-leads/config', {
   294→          method: 'PUT',
   295→          headers: { 'Content-Type': 'application/json' },
   296→          body: JSON.stringify({ enabled: pollEnabled, formIds: validIds }),
   297→        });
   298→        if (saveRes.ok) {
   299→          setPollSavedEnabled(pollEnabled);
   300→          setPollFormIds(validIds.length ? validIds : ['']);
   301→        } else {
   302→          const saveData = await saveRes.json();
   303→          toast.error(`Erro ao salvar: ${saveData.error || 'desconhecido'}`);
   304→          return;
   305→        }
   306→      } catch {
   307→        toast.error('Erro ao salvar configuração');
   308→        return;
   309→      }
   310→    }
   311→
   312→    setPollTriggering(true);
   313→    try {
   314→      const res = await fetch('/api/cron/fetch-meta-leads');
   315→      const data = await res.json();
   316→      if (res.status === 401) {
   317→        toast.error('Sessão expirada. Faça login novamente.');
   318→      } else if (res.ok) {
   319→        if (data.status === 'disabled') {
   320→          toast.warning('Polling está desativado. Ative primeiro e salve.');
   321→        } else if (data.status === 'idle') {
   322→          toast.info('Nenhum form ID configurado. Adicione ao menos um form ID.');
   323→        } else {
   324→          toast.success(`Polling executado: ${data.totalFetched} encontrados, ${data.totalImported} importados (${data.elapsed}).`);
   325→          // Atualizar resultado exibido
   326→          setPollLastResult(data);
   327→          setPollLastRun(new Date().toISOString());
   328→        }
   329→      } else {
   330→        toast.error(data.error || 'Erro ao executar polling');
   331→      }
   332→    } catch {
   333→      toast.error('Erro ao executar polling manual');
   334→    } finally {
   335→      setPollTriggering(false);
   336→    }
   337→  }
